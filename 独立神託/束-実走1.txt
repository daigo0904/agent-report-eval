    """値段の文字列を整数（円）にする。

    - 前後の空白は無視する（" 12 " → 12）
    - 先頭の「¥」、末尾の「円」はあってもよい（"¥1,234" → 1234、"1,234円" → 1234）
    - 3桁ごとのカンマはあってもよい。ただし位置が正しくないカンマは ValueError
      （"12,345,678" → 12345678、"1,23" → ValueError）
    - 全角数字も読む（"１２３" → 123）
    - 先頭に「-」があれば負（"-¥5" → -5、"-5円" → -5）
    - 数字が1つも無いもの（""、"¥"、"abc"）は ValueError
    """
    # 1. 前後の空白を削除
    s = s.strip()

    # 2. 全角数字を半角に変換
    s = s.translate(str.maketrans('０１２３４５６７８９', '0123456789'))

    # 3. 符号の判定
    is_negative = False
    if s.startswith('-'):
        is_negative = True
        s = s[1:]

    # 4. 「¥」と「円」の除去
    if s.startswith('¥'):
        s = s[1:]
    if s.endswith('円'):
        s = s[:-1]

    # 5. 数字が含まれているかチェック
    if not any(c.isdigit() for c in s):
        raise ValueError("No digits found in price string")

    # 6. カンマの検証
    parts = s.split(',')
    
    if len(parts) > 1:
        # 最初のパーツ（千の位より上）は、1-3桁であるべき。
        if len(parts[0]) == 0 or len(parts[0]) > 3:
             raise ValueError("Invalid comma placement")
        
        # 2番目以降のパーツは、必ず3桁である必要がある。
        for part in parts[1:]:
            if len(part) != 3:
                raise ValueError("Invalid comma placement")
    
    # カンマを除去して、残りが数字のみか確認
    clean_s = s.replace(',', '')
    if not clean_s.isdigit():
        raise ValueError("Invalid characters in price string")

    result = int(clean_s)
    return -result if is_negative else result

from price import parse_price

def test_parse_price():
    # Basic cases
    assert parse_price('1234') == 1234
    assert parse_price('  12  ') == 12
    
    # Currency symbols
    assert parse_price('¥1,234') == 1234
    assert parse_price('1,234円') == 1234
    assert parse_price('¥1234円') == 1234
    
    # Commas
    assert parse_price('12,345,678') == 12345678
    try:
        parse_price('1,23')
        assert False, "Should have raised ValueError for incorrect comma placement"
    except ValueError:
        pass
    
    # Full-width numbers
    assert parse_price('１２３') == 123
    
    # Negative numbers
    assert parse_price('-¥5') == -5
    assert parse_price('-5円') == -5
    assert parse_price('-1,234') == -1234
    
    # Error cases
    for invalid in ['', '¥', 'abc', '1,2,3', '12,34,567']: # Wait, 12,34,567 is actually wrong too
        # Actually, the docstring says "3桁ごとのカンマはあってもよい。ただし位置が正しくないカンマは ValueError"
        # "1,2,3" is definitely wrong.
        pass

    # Let's refine the error cases test
    for invalid in ['', '¥', 'abc']:
        try:
            parse_price(invalid)
            assert False, f"Should have raised ValueError for {invalid}"
        except ValueError:
            pass

    print('All tests passed!')

if __name__ == "__main__":
