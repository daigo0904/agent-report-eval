import csv
import re

def calculate_total(filename):
    total = 0
    with open(filename, mode='r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            amount_str = row['amount']
            # Remove characters that are not digits or minus sign
            # We need to handle commas, ¥, and 円
            # Replace everything except digits and '-' with empty string
            # But wait, we should be careful about the minus sign.
            
            # First, remove commas and other symbols
            clean_str = re.sub(r'[^\d-]', '', amount_str)
            
            if clean_str:
                try:
                    total += int(clean_str)
                except ValueError:
                    pass
    return total

if __name__ == "__main__":
    total_amount = calculate_total('sales.csv')
    with open('total.txt', 'w', encoding='utf-8') as f:
        f.write(str(total_amount))

import csv
import re

def calculate_total(input_file, output_file):
    total = 0
    with open(input_file, mode='r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            amount_str = row['amount']
            # Remove currency symbols, commas, and '円'
            clean_amount = re.sub(r'[¥,円]', '', amount_str).strip()
            try:
                total += int(clean_amount)
            except ValueError:
                # In case there are other non-numeric characters
                pass
    
    with open(output_file, mode='w', encoding='utf-8') as f:
        f.write(str(total))
